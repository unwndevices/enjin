```cpp
#ifdef ESP32
void LuaPlatform::openEmbeddedLibraries(lua_State* L) {
    // ESP32: Open only essential libraries to save memory
    luaL_requiref(L, "_G", luaopen_base, 1);
    lua_pop(L, 1);
    luaL_requiref(L, LUA_MATHLIBNAME, luaopen_math, 1);
    lua_pop(L, 1);
    luaL_requiref(L, LUA_STRLIBNAME, luaopen_string, 1);
    lua_pop(L, 1);
    luaL_requiref(L, LUA_TABLIBNAME, luaopen_table, 1);
    lua_pop(L, 1);
    // ... conditional UTF8 ...
}
```

From lualib.h (standard Lua 5.x):
```cpp
LUALIB_API int (luaopen_coroutine) (lua_State *L);
#define LUA_COLIBNAME   "coroutine"
```

Note: LuaJIT bundles coroutines in luaopen_base — luaopen_coroutine is NOT separately declared in LuaJIT headers. This code is inside #ifdef ESP32 which uses standard Lua, so luaopen_coroutine is available.
