       LuaBindings* b = getBindings(L);
       if (!b || !b->currentInput) { lua_pushboolean(L, 0); return 1; }
       int btn = static_cast<int>(luaL_checkinteger(L, 1));
       lua_pushboolean(L, b->currentInput->held(btn) ? 1 : 0);
       return 1;
   }

   int LuaBindings::lua_isButtonJustPressed(lua_State* L) {
       LuaBindings* b = getBindings(L);
       if (!b || !b->currentInput) { lua_pushboolean(L, 0); return 1; }
       int btn = static_cast<int>(luaL_checkinteger(L, 1));
       lua_pushboolean(L, b->currentInput->justPressed(btn) ? 1 : 0);
       return 1;
   }

   int LuaBindings::lua_isButtonJustReleased(lua_State* L) {
       LuaBindings* b = getBindings(L);
       if (!b || !b->currentInput) { lua_pushboolean(L, 0); return 1; }
       int btn = static_cast<int>(luaL_checkinteger(L, 1));
       lua_pushboolean(L, b->currentInput->justReleased(btn) ? 1 : 0);
       return 1;
   }

   int LuaBindings::lua_getAxis(lua_State* L) {
       LuaBindings* b = getBindings(L);
       if (!b || !b->currentInput) { lua_pushnumber(L, 0.0); return 1; }
       int axis = static_cast<int>(luaL_checkinteger(L, 1));
       float val = (axis >= 0 && axis < 8) ? b->currentInput->axes[axis] : 0.0f;
       lua_pushnumber(L, static_cast<lua_Number>(val));
       return 1;
   }

**Script content:**

```lua
-- scripts/e2e_parity.lua
-- E2E parity test: 15-color palette grid + button-0 input indicator.
-- Runs identically on SDL3, WASM, and ESP32.
-- Host calls update(dt) and draw() each frame.

local COLS   = 5
local CELL_W = 24
local CELL_H = 24

local function draw_color_grid()
    -- 5x3 grid: 15 cells covering palette indices 0-14.
    -- Positioned at top-left (0, 0) — occupies 120x72 canvas pixels.
    for i = 0, 14 do
        local col = i % COLS
        local row = math.floor(i / COLS)
        setColor(i)
        rectangle(col * CELL_W, row * CELL_H, CELL_W, CELL_H)
    end
end

local function draw_input_indicator()
    -- 24x24 cell in the bottom-right corner of the 128x128 canvas.
    -- Color index 7 (bright) when button 0 held; index 2 (dim) when not held.
    local x = getWidth() - CELL_W
    local y = getHeight() - CELL_H
    if isButtonHeld(0) then
        setColor(7)
    else
        setColor(2)
    end
    rectangle(x, y, CELL_W, CELL_H)
end

function update(dt)
    -- No per-frame state to update.
    -- Input is polled by the host and written to InputState before this call.
end

function draw()
    clear(0)
    draw_color_grid()
    draw_input_indicator()
end
```

**Notes:**
- `rectangle(x, y, w, h)` with no mode string defaults to "fill" in the existing `lua_rectangle` binding (confirmed in bindings.cpp). This is correct — do not add a mode string.
- `getWidth()` / `getHeight()` are already bound — they return canvas dimensions.
- `clear(0)` clears to palette index 0 (typically black in the default palette).
- Button index 0 = BTN_UP (arrow up / W key) per the BTN_UP enum in sdl_main.cpp. The script documentation does not need to reference this — index 0 is sufficient.
- The `scripts/` directory must be created at the project root if it does not already exist. Check with `ls scripts/ 2>/dev/null` first; use Write tool to create the file.
