<task type="auto">
  <name>Add CMake docs target for doxygen generation</name>
  <files>CMakeLists.txt</files>
  <action>
    1. Read existing CMakeLists.txt to understand project structure
    2. Add Doxygen finding and docs target configuration:
       find_package(Doxygen)
       if(DOXYGEN_FOUND)
           set(DOXYGEN_IN ${CMAKE_CURRENT_SOURCE_DIR}/docs/Doxyfile)
           set(DOXYGEN_OUT ${CMAKE_CURRENT_BINARY_DIR}/docs)

           add_custom_target(docs
               COMMAND ${DOXYGEN_EXECUTABLE} ${DOXYGEN_IN}
               WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
               COMMENT "Generating API documentation with Doxygen"
               BYPRODUCTS ${DOXYGEN_OUT}/xml/index.xml
           )

           # Make docs depend on enjin2 library targets
           add_dependencies(docs enjin2)
       endif()

    Reference: .planning/phases/06-create-library-docs-using-doxygen-docusaurus/06-RESEARCH.md (CMake Integration section)

    Note: Verify the target name for enjin2 library (may be enjin2_core, enjin2, or similar).
  </action>
  <verify>grep "add_custom_target(docs" CMakeLists.txt</verify>
  <done>CMakeLists.txt includes find_package(Doxygen) and docs target</done>
</task>
