<task type="auto">
  <name>Configure Doxygen and CMake</name>
  <files>docs/Doxyfile, CMakeLists.txt</files>
  <action>
    1. Create docs directory: mkdir -p docs

    2. Create docs/Doxyfile with minimal XML generation configuration:

    ```makefile
    # Project identification
    PROJECT_NAME           = "enjin2"
    PROJECT_NUMBER         = "2.0.0"
    PROJECT_BRIEF          = "Lightweight C++ game engine"

    # Output settings
    OUTPUT_DIRECTORY       = docs/xml
    GENERATE_XML          = YES
    GENERATE_HTML         = NO
    XML_OUTPUT             = xml

    # Input settings
    INPUT                  = include/enjin2
    FILE_PATTERNS         = *.hpp
    RECURSIVE             = YES
    EXCLUDE_PATTERNS       = */test/* */tests/* */examples/*

    # Extraction settings
    EXTRACT_ALL            = NO   # Only document what has comments
    EXTRACT_PRIVATE        = NO
    EXTRACT_STATIC         = NO
    EXTRACT_LOCAL_CLASSES  = NO

    # Warning settings
    WARN_IF_UNDOCUMENTED   = YES
    WARN_IF_DOC_ERROR      = YES
    WARN_NO_PARAMDOC       = NO   # Allow undocumented params temporarily

    # C++ specific
    OPTIMIZE_OUTPUT_JAVA   = NO   # Use C++ settings
    BUILTIN_STL_SUPPORT    = YES   # Better std:: documentation

    # Preprocessor
    ENABLE_PREPROCESSING   = YES
    MACRO_EXPANSION        = YES
    EXPAND_ONLY_PREDEF     = YES

    # Graphs (disabled for performance, enable if needed)
    HAVE_DOT               = NO
    CALL_GRAPH             = NO
    CALLER_GRAPH           = NO
    CLASS_GRAPH            = NO
    ```

    3. Add documentation generation target to root CMakeLists.txt after line 36 (after project(engin2...)):

    ```cmake
    # Documentation generation
    find_package(Doxygen)
    if(DOXYGEN_FOUND)
        set(DOXYGEN_IN ${CMAKE_CURRENT_SOURCE_DIR}/docs/Doxyfile)
        set(DOXYGEN_OUT ${CMAKE_CURRENT_SOURCE_DIR}/docs/xml)

        add_custom_target(docs
            COMMAND ${DOXYGEN_EXECUTABLE} ${DOXYGEN_IN}
            WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
            COMMENT "Generating API documentation with Doxygen"
            BYPRODUCTS ${DOXYGEN_OUT}/index.xml
        )

        message(STATUS "Doxygen found - documentation target available: cmake --build . --target docs")
    else()
        message(WARNING "Doxygen not found - documentation generation unavailable")
    endif()
    ```

    This creates a 'docs' target that can be built with cmake --build . --target docs

    Reference: 06-RESEARCH.md "Doxygen Configuration for XML Output" section
