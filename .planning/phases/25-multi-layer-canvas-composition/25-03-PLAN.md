void LuaBindings::setLayers(LuaCanvas** canvases, uint8_t count, bool* visibleArr) {
    layerCount = (count > MAX_LUA_LAYERS) ? MAX_LUA_LAYERS : count;
    for (uint8_t i = 0; i < layerCount; ++i) {
        layerCanvases[i] = canvases[i];
    }
    layerVisible = visibleArr;
    activeLayer = 0;
    currentCanvas = layerCanvases[0];
}
```

2. **Implement 6 lua_CFunction static methods** (add a new section after sprite pool bindings):

```
//==============================================================================
// Layer System Bindings (LAYER-06)
//==============================================================================
```

**lua_setLayer(L):** Takes 1 arg (Lua 1-indexed layer number). Convert to 0-based C++ index: `cpp_idx = lua_idx - 1`. Clamp to `[0, layerCount-1]` (silent clamp, matching setFrame pattern). Set `b->activeLayer = cpp_idx` and `b->currentCanvas = b->layerCanvases[cpp_idx]`. Return 0.

**lua_getLayer(L):** Returns `b->activeLayer + 1` as Lua integer (1-indexed). If no bindings, push 0.

**lua_clearLayer(L):** Takes 2 args: layer (1-indexed), color (palette index). Convert layer to 0-based, clamp. Call `b->layerCanvases[cpp_idx]->clear(color)`. Return 0. If only 1 arg, use index 0 (black) as default clear color. If layer canvas is null, return 0 (no-op).

**lua_getLayerCount(L):** Push `b->layerCount` as integer. Return 1.

**lua_setLayerVisible(L):** Takes 2 args: layer (1-indexed), visible (boolean). Convert layer to 0-based, clamp. If `b->layerVisible` is not null, set `b->layerVisible[cpp_idx] = lua_toboolean(L, 2)`. Return 0.

**lua_isLayerVisible(L):** Takes 1 arg: layer (1-indexed). Convert to 0-based, clamp. If `b->layerVisible` not null, push `b->layerVisible[cpp_idx]` as boolean. Else push true. Return 1.

-- layer_demo.lua — Multi-layer composition demo for enjin2
-- Draws different content on each layer to verify compositor.

function update(dt)
    -- no-op for this demo
end

function draw()
    -- Layer 1 (BG): fill with dark blue (palette index 1)
    setLayer(LAYER_BG)
    clear(1)

    -- Layer 2 (MID): draw a green rectangle in the center
    setLayer(LAYER_MID)
    setColor(11)  -- green
    rectangle("fill", 32, 32, 64, 64)

    -- Layer 3 (FG): draw a red circle overlapping the rectangle
    setLayer(LAYER_FG)
    setColor(8)  -- red
    circle("fill", 64, 64, 24)

    -- Layer 4 (UI): draw score text area at top
    setLayer(LAYER_UI)
    setColor(7)  -- white
    rectangle("fill", 0, 0, 128, 12)
    setColor(0)  -- black
    -- Draw "SCORE" as pixel dots (simplified)
    for i = 0, 4 do
        point(4 + i * 2, 4)
    end

    -- Report layer info
    local count = getLayerCount()
    local current = getLayer()
    print("Layers: " .. count .. ", current: " .. current)
end
```
