template<typename T>
size_t findObjects(T** results, size_t maxResults) {
    // caller provides buffer; function returns count written
    size_t found = 0;
    for (size_t i = 0; i < objectCount && found < maxResults; ++i) {
        if (auto obj = dynamic_cast<T*>(objects[i].get())) {
            results[found++] = obj;
        }
    }
    return found;
}

      getTagCount() after 2     -> 2
      adding 8 tags             -> all 8 present, addTag returns true for each

      9th tag not findable      -> hasTag("tag9") == false
      clearTags()               -> getTagCount() == 0, hasTag("enemy") == false

    OBJ-04: ObjectCollection::findAllWithTag
      two objects tagged "enemy", one tagged "item" -> findAllWithTag("enemy", buf, 10) returns 2
      buffer receives both enemy Object* pointers
      findAllWithTag("item", buf, 10)   -> returns 1
      findAllWithTag("ghost", buf, 10)  -> returns 0
      findAllWithTag with maxResults=1  -> returns only 1 even if 2 match
  </behavior>
