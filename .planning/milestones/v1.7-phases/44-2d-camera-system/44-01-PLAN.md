template<typename PixelType>
void renderObjects(ICanvas<PixelType>& canvas) {
    static constexpr size_t MAX_DRAWABLES = 256;
    C_Drawable* drawables[MAX_DRAWABLES];
    size_t drawableCount = 0;
    objects.forEach([&](Object* obj) {
        if (!obj || !obj->isActive()) return;
        static constexpr size_t OBJ_MAX_DRAW = 16;
        C_Drawable* objDrawables[OBJ_MAX_DRAW];
        size_t n = obj->getComponents<C_Drawable>(objDrawables, OBJ_MAX_DRAW);
        for (size_t i = 0; i < n && drawableCount < MAX_DRAWABLES; ++i) {
            if (objDrawables[i]->isVisible()) {
                drawables[drawableCount++] = objDrawables[i];
            }
        }
    });
    std::sort(drawables, drawables + drawableCount,
              [](const C_Drawable* a, const C_Drawable* b) {
                  return a->shouldDrawBefore(*b);
              });
    for (size_t i = 0; i < drawableCount; ++i) {
        if constexpr (std::is_same_v<PixelType, Pixel4>) {
            drawables[i]->draw(canvas);
        } else {
            (void)i;
        }
    }
}

**1. Create `include/enjin2/components/camera.hpp`:**

```cpp
#pragma once
#include "../core/component.hpp"
#include "../core/types.hpp"
#include <cmath>
#include <algorithm>

namespace enjin2 {

class C_Camera : public Component {
public:
    explicit C_Camera(Object* owner);
    ~C_Camera() override = default;

    void update(float dt) override;

    // Position (float for smooth sub-pixel)
    void setPosition(float x, float y);
    Vec2 getPosition() const { return m_pos; }

    // Smooth follow: sets target and lerp speed (0 = instant snap via setPosition, 0.1 = smooth)
    void lookAt(float x, float y, float lerpSpeed = 1.0f);

    // Screen shake: sin oscillation with decay
    void shake(float intensity, float duration);

    // Integer screen offset for render pipeline: -(m_pos + m_shakeOffset)
    // Convention: camera pos = world point at top-left of screen (simple subtract)
    Point getScreenOffset() const;

    // Viewport clamping (0,0,0,0 = unclamped)
    void setBounds(float minX, float minY, float maxX, float maxY);
    void clearBounds();
    bool hasBounds() const { return m_hasBounds; }

    // Canvas dimensions (injected by host or via setter, defaults to 128x128)
    void setCanvasSize(uint16_t w, uint16_t h) { m_canvasW = w; m_canvasH = h; }
    uint16_t getCanvasWidth() const { return m_canvasW; }
    uint16_t getCanvasHeight() const { return m_canvasH; }

private:
    Vec2 m_pos{0.f, 0.f};
    Vec2 m_target{0.f, 0.f};
    float m_lerpSpeed{1.0f};      // 1.0 = instant, <1 = smooth

    // Shake
    float m_shakeIntensity{0.f};
    float m_shakeDuration{0.f};
    float m_shakeElapsed{0.f};
    Vec2  m_shakeOffset{0.f, 0.f};

    // Bounds
    bool  m_hasBounds{false};
    float m_boundsMinX{0.f}, m_boundsMinY{0.f};
    float m_boundsMaxX{0.f}, m_boundsMaxY{0.f};

    // Canvas dims for offset calc
    uint16_t m_canvasW{128};
    uint16_t m_canvasH{128};

    void clampPosition();
};

} // namespace enjin2
```

**2. Create `src/components/camera.cpp`:**

Implement:
- Constructor: calls Component(owner), all members initialized via in-class defaults.
- `setPosition(x, y)`: sets m_pos AND m_target to (x,y), m_lerpSpeed=1.0 (instant — no lerp residual). Calls clampPosition().
- `lookAt(x, y, lerpSpeed)`: sets m_target=(x,y), m_lerpSpeed=lerpSpeed. If lerpSpeed >= 1.0, also sets m_pos = m_target immediately.
- `update(float dt)`: (a) Lerp m_pos toward m_target: `m_pos += (m_target - m_pos) * std::min(m_lerpSpeed * dt * 10.0f, 1.0f);` (scale by 10 so lerpSpeed=0.1 gives visually smooth tracking). (b) Clamp position. (c) Update shake: if m_shakeElapsed < m_shakeDuration, compute `float progress = m_shakeElapsed / m_shakeDuration; float decay = 1.0f - progress; m_shakeOffset.x = std::sin(m_shakeElapsed * 40.0f) * m_shakeIntensity * decay; m_shakeOffset.y = std::sin(m_shakeElapsed * 30.0f) * m_shakeIntensity * decay * 0.7f;` then `m_shakeElapsed += dt;`. Else set m_shakeOffset to (0,0).
- `shake(intensity, duration)`: sets m_shakeIntensity, m_shakeDuration, resets m_shakeElapsed to 0.
- `getScreenOffset()`: returns `Point(static_cast<int16_t>(-(m_pos.x + m_shakeOffset.x)), static_cast<int16_t>(-(m_pos.y + m_shakeOffset.y)));`. Convention: camera pos is the world point at the top-left of the screen. Screen pos = world pos + screenOffset = world pos - camera pos.
- `setBounds(minX, minY, maxX, maxY)`: stores bounds, sets m_hasBounds=true.
- `clearBounds()`: sets m_hasBounds=false.
- `clampPosition()`: if m_hasBounds, clamp m_pos.x to [m_boundsMinX, m_boundsMaxX], m_pos.y to [m_boundsMinY, m_boundsMaxY] using std::max/std::min.

  ```cpp
  /// @brief Set screen-space mode. Screen-space drawables skip camera offset.
  void setScreenSpace(bool ss) { m_screenSpace = ss; }
  /// @brief Check if drawable is in screen-space mode
  bool isScreenSpace() const { return m_screenSpace; }
  /// @brief Draw with camera offset applied. Screen-space drawables ignore offset.
  virtual void drawWithOffset(ICanvas<Pixel4>& canvas, Point offset) {
      if (m_screenSpace) {
          draw(canvas);
          return;
      }
      Point saved = anchor_offset;
      anchor_offset += offset;   // offset is negative camera pos, so adding moves drawable left/up
      draw(canvas);
      anchor_offset = saved;
  }
  ```
  Note on sign convention: `getScreenOffset()` returns negative camera position. Adding it to anchor_offset shifts the drawable in the correct direction (world pos - camera pos = screen pos). Verify by testing: camera at (10,0), drawable at world (50,0), screen pos should be (40,0). offset = Point(-10, 0), anchor_offset starts at (0,0), after += offset: anchor_offset = (-10, 0). GetOffsetPosition() returns position + anchor_offset. If position.x = 50, result = 50 + (-10) = 40. Correct.
