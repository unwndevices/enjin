**Create `include/enjin2/components/tilemap.hpp`:**

```cpp
#pragma once
#include "drawable.hpp"
#include "../graphics/sprite.hpp"

namespace enjin2 {

class C_Tilemap : public C_Drawable {
public:
    static constexpr uint8_t MAX_MAP_W = 64;
    static constexpr uint8_t MAX_MAP_H = 64;

    // Constructor: owner, initial map width/height (can be 0, set later via setTiles)
    C_Tilemap(Object* owner);

    // Tileset
    void setSheet(const SpriteSheet& sheet);
    const SpriteSheet& getSheet() const { return m_sheet; }

    // Map data: copies w*h bytes from data into m_tiles, sets dimensions
    void setTiles(const uint8_t* data, uint8_t w, uint8_t h);

    // Per-tile get/set (0-indexed grid coords, bounds-checked)
    void setTile(uint8_t tx, uint8_t ty, uint8_t tileId);
    uint8_t getTile(uint8_t tx, uint8_t ty) const;

    // Dimensions
    uint8_t getMapWidth() const { return m_mapW; }
    uint8_t getMapHeight() const { return m_mapH; }

    // Camera offset (tilemap-scoped, not engine-wide)
    void setScroll(int16_t sx, int16_t sy);
    int16_t getScrollX() const { return m_scrollX; }
    int16_t getScrollY() const { return m_scrollY; }

    // Coordinate helpers
    // pixelToTile: world pixel -> grid coord (accounts for scroll)
    void pixelToTile(int16_t px, int16_t py, int16_t& tx, int16_t& ty) const;
    // tileToPixel: grid coord -> world pixel top-left
    void tileToPixel(int16_t tx, int16_t ty, int16_t& px, int16_t& py) const;
    // tileAtPixel: returns tile ID at world pixel position (for collision)
    uint8_t tileAtPixel(int16_t px, int16_t py) const;

    // C_Drawable overrides
    void draw(ICanvas<Pixel4>& canvas) override;
    bool continueToDraw() const override;

private:
    uint8_t     m_tiles[MAX_MAP_W * MAX_MAP_H];  // 4096 bytes, zero-alloc
    SpriteSheet m_sheet;
    uint8_t     m_mapW{0};
    uint8_t     m_mapH{0};
    int16_t     m_scrollX{0};
    int16_t     m_scrollY{0};
};

} // namespace enjin2
```

**Constructor** takes only `owner` (no width/height params since tilemap dimensions are set via setTiles). Calls `C_Drawable(owner, 0, 0)` — the drawable width/height are not meaningful for tilemap (it manages its own viewport). Sets `buffer_index = 0` (layer 0 = background per v1.4 convention). Zero-initializes `m_tiles` with `memset(m_tiles, 0, sizeof(m_tiles))`.

**Create `src/components/tilemap.cpp`:**

Implement all methods:

- `setSheet()`: Copy SpriteSheet value.
- `setTiles(data, w, h)`: Clamp w/h to MAX_MAP_W/MAX_MAP_H. Zero entire m_tiles array first, then copy `w*h` bytes from data into m_tiles row by row (respecting the MAX_MAP_W stride). Set m_mapW, m_mapH.
- `setTile(tx, ty, id)`: Bounds check `tx < m_mapW && ty < m_mapH`. Set `m_tiles[ty * MAX_MAP_W + tx] = id`. Use MAX_MAP_W as stride (not m_mapW) so the internal layout is consistent.

  **IMPORTANT stride decision:** Use `m_mapW` as the stride, NOT `MAX_MAP_W`. The tile array is logically `m_mapW * m_mapH` elements stored contiguously from index 0. This matches how `setTiles()` copies data and how Lua's flat table maps to indices. Using MAX_MAP_W stride would waste memory gaps and break flat-table round-tripping. So: `m_tiles[ty * m_mapW + tx]`.

- `getTile(tx, ty)`: Bounds check, return `m_tiles[ty * m_mapW + tx]` or 0 if out of bounds.
- `setScroll(sx, sy)`: Store values directly (no clamping at set time — clamping is in draw()).
- `pixelToTile(px, py, &tx, &ty)`: `tx = (px + m_scrollX) / tileW; ty = (py + m_scrollY) / tileH;` — Note: px/py are screen coordinates, so add scroll to get world position, then divide by tile size. Handle negative values with integer floor division (C++ truncation toward zero needs correction for negatives).
- `tileToPixel(tx, ty, &px, &py)`: `px = tx * tileW - m_scrollX; py = ty * tileH - m_scrollY;` — converts grid coord to screen pixel position.
- `tileAtPixel(px, py)`: Calls pixelToTile, bounds checks, returns getTile or 0 if out of bounds.

- **`draw(ICanvas<Pixel4>& canvas)` — viewport culled rendering:**
  ```
  if (!is_visible || !m_sheet.data || m_mapW == 0 || m_mapH == 0) return;
  const int16_t tileW = static_cast<int16_t>(m_sheet.cellW);
  const int16_t tileH = static_cast<int16_t>(m_sheet.cellH);
  if (tileW == 0 || tileH == 0) return;
  const int16_t canvasW = static_cast<int16_t>(canvas.getWidth());
  const int16_t canvasH = static_cast<int16_t>(canvas.getHeight());

  // First visible tile (clamp negatives to 0)
  int16_t startTX = m_scrollX / tileW;
  int16_t startTY = m_scrollY / tileH;
  if (startTX < 0) startTX = 0;
  if (startTY < 0) startTY = 0;

  // Last visible tile (exclusive), +2 for partial tiles at edges
  int16_t endTX = (m_scrollX + canvasW) / tileW + 1;
  int16_t endTY = (m_scrollY + canvasH) / tileH + 1;
  if (endTX > m_mapW) endTX = m_mapW;
  if (endTY > m_mapH) endTY = m_mapH;

  for (int16_t ty = startTY; ty < endTY; ++ty) {
      for (int16_t tx = startTX; tx < endTX; ++tx) {
          uint8_t tileId = m_tiles[ty * m_mapW + tx];
          if (tileId == 0) continue;  // transparent sentinel
          int16_t px = tx * tileW - m_scrollX;
          int16_t py = ty * tileH - m_scrollY;
          m_sheet.draw(canvas, tileId, px, py);
          // Tile ID used directly as frameIndex: tile 0 = skip, tile 1 = frame 1, etc.
          // Tileset frame 0 is "wasted" (never referenced by drawn tiles).
          // This avoids off-by-one subtraction in the hot path.
      }
  }
  ```

- `continueToDraw()`: Return `!owner->isQueuedForRemoval()` (same as C_Sprite).
